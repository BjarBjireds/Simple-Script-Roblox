-- LOAD DICTIONARY + BUCKET

local WordList = {}
local Buckets = {}

local success, result = pcall(function()
    local raw = game:HttpGet(rawGithubUrl)
    local list = {}
    for line in raw:gmatch("[^\r\n]+") do
        local w = line:match("^%s*(.-)%s*$")
        if w and #w >= 3 and w:match("^[A-Za-z]+$") then
            table.insert(list, string.upper(w))
        end
    end
    return list
end)

if success and type(result) == "table" and #result > 0 then
    WordList = result
    print("[CLOUD] Loaded", #WordList, "words")
    
    for _, w in ipairs(WordList) do
        local c1 = w:sub(1,1)
        Buckets[c1] = Buckets[c1] or {}
        table.insert(Buckets[c1], w)
        
        if #w >= 2 then
            local c2 = w:sub(1,2)
            Buckets[c2] = Buckets[c2] or {}
            table.insert(Buckets[c2], w)
        end
        if #w >= 3 then
            local c3 = w:sub(1,3)
            Buckets[c3] = Buckets[c3] or {}
            table.insert(Buckets[c3], w)
        end
    end
    print("[CLOUD] Buckets ready")
else
    warn("[ERROR] Gagal load dictionary!")
end

-- ==========================================
-- KIRIM & KETIK
-- ==========================================
local function SendKey(char)
    if Remote then pcall(function() Remote:InvokeServer("keyStroke", string.upper(char)) end) end
end

local function SendEnter()
    if Remote then pcall(function() Remote:InvokeServer("tryAnswer") end) end
end

-- FIX: Mengetik mulai dari huruf yang kosong saja
local function AutoTypeWord(word, prefixLength)
    word = string.upper(word)
    local startIdx = prefixLength + 1
    
    if startIdx <= #word then
        for i = startIdx, #word do
            SendKey(word:sub(i, i))
            task.wait(0.025)
        end
    end
    task.wait(0.06)
    SendEnter()
end

-- ==========================================
-- GET WORDS
-- ==========================================
local function GetMatchingWords(prefix)
    if not prefix or prefix == "" then return {} end
    prefix = string.upper(prefix)
    
    local source = Buckets[prefix] or Buckets[prefix:sub(1,1)] or {}
    local list = {}
    
    for _, w in ipairs(source) do
        if w:sub(1, #prefix) == prefix then
            table.insert(list, w)
            if #list >= 600 then break end
        end
    end
    return list
end

-- ==========================================
-- RENDER
-- ==========================================
local function SortAndDisplay(mode, forcePrefix)
    local prefix = forcePrefix or activePrefix
    currentSortMode = mode or currentSortMode or "default"

    local words = GetMatchingWords(prefix)

    for _, child in pairs(ResultScroll:GetChildren()) do
        if child:IsA("TextButton") or child:IsA("TextLabel") then
            child:Destroy()
        end
    end

    if #words == 0 then
        local err = Instance.new("TextLabel", ResultScroll)
        err.Size = UDim2.new(1, -10, 0, 25)
        err.BackgroundTransparency = 1
        err.TextColor3 = Color3.fromRGB(200, 80, 80)
        err.Text = (not prefix or prefix == "") and "Menunggu awalan..." or ("Tidak ada kata untuk " .. prefix)
        err.Font = Enum.Font.SourceSansItalic
        err.TextSize = 13
        ResultScroll.CanvasSize = UDim2.new(0, 0, 0, 30)
        return
    end

    if mode == "short" then
        table.sort(words, function(a, b) if #a == #b then return a < b end return #a < #b end)
    elseif mode == "longest" then
        table.sort(words, function(a, b) if #a == #b then return a < b end return #a > #b end)
    elseif mode == "random" then
        for i = #words, 2, -1 do
            local j = math.random(1, i)
            words[i], words[j] = words[j], words[i]
        end
    else
        table.sort(words)
    end

    local displayCount = math.min(#words, MAX_DISPLAY)
    for i = 1, displayCount do
        local word = words[i]
        local ansBtn = Instance.new("TextButton", ResultScroll)
        ansBtn.Size = UDim2.new(1, -10, 0, 26)
        ansBtn.BackgroundColor3 = Color3.fromRGB(25, 70, 100)
        ansBtn.TextColor3 = Color3.fromRGB(100, 220, 255)
        ansBtn.Text = " > " .. word
        ansBtn.Font = Enum.Font.SourceSansBold
        ansBtn.TextSize = 14
        ansBtn.TextXAlignment = Enum.TextXAlignment.Left

        ansBtn.Activated:Connect(function()
            if isAutoTypeOn then
                ansBtn.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
                ansBtn.TextColor3 = Color3.fromRGB(0, 0, 0)
                AutoTypeWord(word, #prefix)
                task.wait(0.1)
                ansBtn.BackgroundColor3 = Color3.fromRGB(25, 70, 100)
                ansBtn.TextColor3 = Color3.fromRGB(100, 220, 255)
            else
                ansBtn.BackgroundColor3 = Color3.fromRGB(150, 60, 60)
                task.wait(0.12)
                ansBtn.BackgroundColor3 = Color3.fromRGB(25, 70, 100)
            end
        end)
    end

    ResultScroll.CanvasSize = UDim2.new(0, 0, 0, displayCount * 28)
end

BtnShort.Activated:Connect(function() SortAndDisplay("short") end)

local function getExactPrefix()
    local prefix = ""
    local pg = LP:FindFirstChild("PlayerGui")
    if pg and pg:FindFirstChild("ScreenGui") and pg.ScreenGui:FindFirstChild("TopBar") then
        local answerInput = pg.ScreenGui.TopBar:FindFirstChild("AnswerInput")
        local keysFolder = answerInput and answerInput:FindFirstChild("Keys")
        
        if keysFolder then
            for i = 1, 99 do 
                local keyNode = keysFolder:FindFirstChild(tostring(i))
                if not keyNode then break end 
                
                -- V1.6 Fix: Cari TextLabel dimanapun dia bersembunyi di dalam kotak hurufnya
                local textLabel = keyNode:FindFirstChildWhichIsA("TextLabel", true)
                
                if textLabel and textLabel.Text ~= "" and textLabel.Text ~= " " then
                    prefix = prefix .. string.upper(textLabel.Text)
                end
            end
        end
    end
    return prefix
end

task.spawn(function()
    while task.wait(RADAR_INTERVAL) do
        pcall(function()
            local foundPrefix = getExactPrefix()
            
            if foundPrefix and foundPrefix ~= "" then
                if foundPrefix ~= activePrefix then
                    activePrefix = foundPrefix
                    LetterStatus.Text = "🎯 Awalan: " .. activePrefix
                    LetterStatus.TextColor3 = Color3.fromRGB(80, 255, 160)
                    SortAndDisplay(currentSortMode, activePrefix)
                end
            else
                if activePrefix ~= "" then
                    activePrefix = ""
                    LetterStatus.Text = "🎯 Awalan: [ Menunggu... ]"
                    LetterStatus.TextColor3 = Color3.fromRGB(255, 220, 50)
                    SortAndDisplay("default", "")
                end
            end
        end)
    end
end)
