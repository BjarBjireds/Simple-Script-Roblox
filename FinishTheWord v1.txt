-- ==========================================
-- ☁️ MENGAMBIL DICTIONARY DARI GITHUB ☁️
-- ==========================================
-- Ganti link di bawah dengan raw GitHub milikmu.
-- Format file harus: return { "APPLE", "BANANA", "CAT", ... }
-- (satu kata per elemen, boleh UPPERCASE atau lowercase)
local rawGithubUrl = "https://raw.githubusercontent.com/BjarBjireds/Simple-Script-Roblox/refs/heads/main/mega_kamus_english.lua" -- ⚠️ GANTI LINK INI ⚠️

local WordList = {}
local success, result = pcall(function()
    return loadstring(game:HttpGet(rawGithubUrl))()
end)

if success and type(result) == "table" then
    -- Normalisasi ke uppercase biar konsisten
    for _, w in ipairs(result) do
        if type(w) == "string" and #w >= 3 then
            table.insert(WordList, string.upper(w))
        end
    end
    print("[CLOUD] Dictionary berhasil dimuat! Total kata:", #WordList)
else
    warn("[ERROR] Gagal memuat Dictionary! Pastikan link raw benar dan format 'return { \"KATA1\", \"KATA2\", ... }'")
end

local activeLetter = ""
local currentSortMode = "default"

-- ==========================================
-- FUNGSI KIRIM KEYSTROKE & ANSWER
-- ==========================================
local function SendKey(char)
    pcall(function()
        Remote:InvokeServer("keyStroke", string.upper(char))
    end)
end

local function SendEnter()
    pcall(function()
        Remote:InvokeServer("tryAnswer")
    end)
end

local function AutoTypeWord(word)
    word = string.upper(word)
    
    -- Ketik setiap huruf
    for i = 1, #word do
        local char = string.sub(word, i, i)
        SendKey(char)
        task.wait(0.03) -- delay kecil biar stabil
    end
    
    task.wait(0.08)
    SendEnter()
end

-- ==========================================
-- AMBIL KATA YANG COCOK
-- ==========================================
local function GetMatchingWords()
    local list = {}
    if activeLetter == "" or #WordList == 0 then return list end
    
    local letter = string.upper(activeLetter)
    for _, word in ipairs(WordList) do
        if string.sub(word, 1, 1) == letter then
            table.insert(list, word)
        end
    end
    return list
end

-- ==========================================
-- SORT + TAMPILKAN
-- ==========================================
local function SortAndDisplay(mode)
    currentSortMode = mode or "default"
    
    local words = GetMatchingWords()
    if #words == 0 then
        -- bersihkan list
        for _, child in pairs(ResultScroll:GetChildren()) do
            if child:IsA("TextButton") or child:IsA("TextLabel") then child:Destroy() end
        end
        local err = Instance.new("TextLabel", ResultScroll)
        err.Size = UDim2.new(1, -10, 0, 25)
        err.BackgroundTransparency = 1
        err.TextColor3 = Color3.fromRGB(200, 80, 80)
        err.Text = activeLetter == "" and "Menunggu huruf..." or "Tidak ada kata untuk huruf " .. activeLetter
        err.Font = Enum.Font.SourceSansItalic
        err.TextSize = 13
        ResultScroll.CanvasSize = UDim2.new(0, 0, 0, 30)
        return
    end

    if mode == "short" then
        table.sort(words, function(a, b)
            if #a == #b then return a < b end
            return #a < #b
        end)
        print("[SORT] SHORT →", #words, "kata")
    elseif mode == "longest" then
        table.sort(words, function(a, b)
            if #a == #b then return a < b end
            return #a > #b
        end)
        print("[SORT] LONGEST →", #words, "kata")
    elseif mode == "random" then
        for i = #words, 2, -1 do
            local j = math.random(1, i)
            words[i], words[j] = words[j], words[i]
        end
        print("[SORT] RANDOM →", #words, "kata")
    else
        table.sort(words)
    end

    -- Render
    for _, child in pairs(ResultScroll:GetChildren()) do
        if child:IsA("TextButton") or child:IsA("TextLabel") then child:Destroy() end
    end

    local foundCount = 0
    for _, word in ipairs(words) do
        foundCount = foundCount + 1
        
        local ansBtn = Instance.new("TextButton", ResultScroll)
        ansBtn.Size = UDim2.new(1, -10, 0, 26)
        ansBtn.BackgroundColor3 = Color3.fromRGB(25, 70, 100)
        ansBtn.TextColor3 = Color3.fromRGB(100, 220, 255)
        ansBtn.Text = " > " .. word
        ansBtn.Font = Enum.Font.SourceSansBold
        ansBtn.TextSize = 14
        ansBtn.TextXAlignment = Enum.TextXAlignment.Left
        
        ansBtn.Activated:Connect(function()
            if isAutoTypeOn then
                ansBtn.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
                ansBtn.TextColor3 = Color3.fromRGB(0, 0, 0)
                AutoTypeWord(word)
                task.wait(0.15)
                ansBtn.BackgroundColor3 = Color3.fromRGB(25, 70, 100)
                ansBtn.TextColor3 = Color3.fromRGB(100, 220, 255)
            else
                ansBtn.BackgroundColor3 = Color3.fromRGB(150, 60, 60)
                task.wait(0.2)
                ansBtn.BackgroundColor3 = Color3.fromRGB(25, 70, 100)
            end
        end)
    end

    ResultScroll.CanvasSize = UDim2.new(0, 0, 0, foundCount * 28)
end

BtnShort.Activated:Connect(function() SortAndDisplay("short") end)
BtnRandom.Activated:Connect(function() SortAndDisplay("random") end)
BtnLongest.Activated:Connect(function() SortAndDisplay("longest") end)

-- ==========================================
-- RADAR DETEKSI HURUF AWAL
-- ==========================================
-- Path dari bahan: PlayerGui > ScreenGui > TopBar > AnswerInput > Keys > 1 > Inner > InnerBack > TextLabel
task.spawn(function()
    while task.wait(0.25) do
        pcall(function()
            local pg = LP:FindFirstChild("PlayerGui")
            if not pg then return end
            
            local screen = pg:FindFirstChild("ScreenGui")
            if not screen then return end
            
            local topBar = screen:FindFirstChild("TopBar")
            if not topBar then return end
            
            local answerInput = topBar:FindFirstChild("AnswerInput")
            if not answerInput then return end
            
            local keys = answerInput:FindFirstChild("Keys")
            if not keys then return end
            
            local key1 = keys:FindFirstChild("1")
            if not key1 then return end
            
            local inner = key1:FindFirstChild("Inner")
            if not inner then return end
            
            local innerBack = inner:FindFirstChild("InnerBack")
            if not innerBack then return end
            
            local textLabel = innerBack:FindFirstChild("TextLabel")
            if not textLabel or not textLabel:IsA("TextLabel") then return end
            
            local raw = textLabel.Text
            local letter = string.match(raw or "", "%a")
            
            if letter then
                letter = string.upper(letter)
                if letter ~= activeLetter then
                    activeLetter = letter
                    LetterStatus.Text = "🎯 Huruf Awal: " .. activeLetter
                    LetterStatus.TextColor3 = Color3.fromRGB(80, 255, 160)
                    SortAndDisplay(currentSortMode)
                end
            else
                if activeLetter ~= "" then
                    activeLetter = ""
                    LetterStatus.Text = "🎯 Huruf Awal: [ Menunggu... ]"
                    LetterStatus.TextColor3 = Color3.fromRGB(255, 220, 50)
                    SortAndDisplay("default")
                end
            end
        end)
    end
end)
